# assets/blender/fishlib.py
"""Shared Blender build steps for scripted fish. Every function is pure bpy; no globals."""
import bpy, bmesh, math, os

def reset_scene():
    bpy.ops.wm.read_factory_settings(use_empty=True)
    s = bpy.context.scene
    s.unit_settings.system = 'METRIC'; s.unit_settings.scale_length = 0.01
    return s

def build_body(name, body_len, body_h, body_w, taper_z=0.55, taper_y=0.4, segments=48, rings=24):
    """UV-sphere scaled to (len, w, h) then tapered toward the tail (-X). Returns the object."""
    bpy.ops.mesh.primitive_uv_sphere_add(segments=segments, ring_count=rings, radius=1.0)
    body = bpy.context.object; body.name = name
    body.scale = (body_len / 2, body_w / 2, body_h / 2)
    bpy.ops.object.transform_apply(scale=True)
    bm = bmesh.new(); bm.from_mesh(body.data)
    for v in bm.verts:
        t = max(0.0, -v.co.x / (body_len / 2))
        v.co.z *= 1.0 - taper_z * t * t
        v.co.y *= 1.0 - taper_y * t
    bm.to_mesh(body.data); bm.free()
    return body

def add_fin(name, verts, faces, thickness=0.25):
    mesh = bpy.data.meshes.new(name); mesh.from_pydata(verts, [], faces); mesh.update()
    ob = bpy.data.objects.new(name, mesh); bpy.context.collection.objects.link(ob)
    sol = ob.modifiers.new("Solidify", 'SOLIDIFY'); sol.thickness = thickness; sol.offset = 0
    return ob

def join_fins(body, fins):
    bpy.ops.object.select_all(action='DESELECT')
    for f in fins: f.select_set(True)
    body.select_set(True); bpy.context.view_layer.objects.active = body
    bpy.ops.object.convert(target='MESH')
    bpy.ops.object.join()
    bpy.ops.object.shade_smooth()
    return body

def unwrap(body):
    bpy.ops.object.select_all(action='DESELECT'); body.select_set(True)
    bpy.context.view_layer.objects.active = body
    bpy.ops.object.mode_set(mode='EDIT'); bpy.ops.mesh.select_all(action='SELECT')
    bpy.ops.uv.smart_project(angle_limit=math.radians(66), island_margin=0.02)
    bpy.ops.object.mode_set(mode='OBJECT')

def bake_base_color(body, mat_name, build_color_nodes, image_name, export_dir, size=2048, samples=16):
    """build_color_nodes(nt, bsdf) must link something into bsdf.inputs['Base Color'].
    Bakes diffuse color to <export_dir>/<image_name>.png and rewires it as base color."""
    mat = bpy.data.materials.new(mat_name)
    if mat.node_tree is None: mat.use_nodes = True     # Blender 5.x may return a material without a node tree
    nt = mat.node_tree; nt.nodes.clear()
    out = nt.nodes.new("ShaderNodeOutputMaterial")
    bsdf = nt.nodes.new("ShaderNodeBsdfPrincipled"); bsdf.inputs["Roughness"].default_value = 0.35
    build_color_nodes(nt, bsdf)
    nt.links.new(bsdf.outputs[0], out.inputs[0])
    body.data.materials.append(mat)
    img = bpy.data.images.new(image_name, size, size)
    tex = nt.nodes.new("ShaderNodeTexImage"); tex.image = img
    for n in nt.nodes: n.select = False
    tex.select = True; nt.nodes.active = tex
    s = bpy.context.scene
    s.render.engine = 'CYCLES'; s.cycles.device = 'CPU'; s.cycles.samples = samples
    s.render.bake.use_pass_direct = False; s.render.bake.use_pass_indirect = False; s.render.bake.use_pass_color = True
    bpy.ops.object.select_all(action='DESELECT'); body.select_set(True); bpy.context.view_layer.objects.active = body
    bpy.ops.object.bake(type='DIFFUSE', margin=8)
    img.filepath_raw = os.path.join(export_dir, image_name + ".png"); img.file_format = 'PNG'; img.save()
    nt.links.new(tex.outputs["Color"], bsdf.inputs["Base Color"])
    return mat

def build_rig(rig_name, half_len, pec_root_y, pec_z=0.5, spine=6, tail_tip_x=None):
    """Root(non-deform) -> Spine0..Spine{spine-1} -> Tail; PecL/PecR under Spine1. Returns armature."""
    bpy.ops.object.armature_add(enter_editmode=True, location=(0, 0, 0))
    arm = bpy.context.object; arm.name = rig_name
    eb = arm.data.edit_bones
    for b in list(eb): eb.remove(b)
    root = eb.new("Root"); root.head = (0, 0, 0); root.tail = (0, 0, 1); root.use_deform = False
    x_head, x_tail = half_len * 0.95, -half_len * 0.9
    step = (x_head - x_tail) / spine
    prev = root
    for i in range(spine):
        b = eb.new(f"Spine{i}"); b.head = (x_head - step * i, 0, 0); b.tail = (x_head - step * (i + 1), 0, 0)
        b.parent = prev; b.use_connect = (i > 0); prev = b
    t = eb.new("Tail"); t.head = prev.tail; t.tail = (tail_tip_x if tail_tip_x is not None else -half_len - 6, 0, 0)
    t.parent = prev; t.use_connect = True
    for name, sign in (("PecL", 1), ("PecR", -1)):
        pb = eb.new(name); pb.head = (half_len * 0.35, sign * pec_root_y, pec_z)
        pb.tail = (half_len * 0.05, sign * (pec_root_y + 4.0), -2.0); pb.parent = eb["Spine1"]
    bpy.ops.object.mode_set(mode='OBJECT')
    return arm

def skin(body, arm):
    bpy.ops.object.select_all(action='DESELECT'); body.select_set(True); arm.select_set(True)
    bpy.context.view_layer.objects.active = arm
    bpy.ops.object.parent_set(type='ARMATURE_AUTO')

def assert_rig_contract(body, arm, spine=6):
    expected = {"Root", "Tail", "PecL", "PecR"} | {f"Spine{i}" for i in range(spine)}
    got = {b.name for b in arm.data.bones}
    assert got == expected, f"bones {sorted(got)} != {sorted(expected)}"
    unweighted = sum(1 for v in body.data.vertices if sum(g.weight for g in v.groups) < 1e-4)
    if unweighted: raise RuntimeError(f"{unweighted} unweighted vertices")
    root_gi = body.vertex_groups["Root"].index if "Root" in body.vertex_groups else -1
    root_max = max((g.weight for v in body.data.vertices for g in v.groups if g.group == root_gi), default=0.0)
    return unweighted, root_max

def save_and_export(body, arm, blend_path, fbx_path):
    bpy.ops.wm.save_as_mainfile(filepath=blend_path)
    bpy.ops.object.select_all(action='DESELECT'); body.select_set(True); arm.select_set(True)
    bpy.ops.export_scene.fbx(filepath=fbx_path, use_selection=True, apply_unit_scale=True,
                             apply_scale_options='FBX_SCALE_NONE', axis_forward='X', axis_up='Z',
                             add_leaf_bones=False, bake_anim=False, mesh_smooth_type='FACE',
                             path_mode='AUTO', embed_textures=False)

def render_preview(path, cam_loc=(20, -45, 10), cam_rot=(1.35, 0, 0.42)):
    s = bpy.context.scene
    cam = bpy.data.objects.new("PreviewCam", bpy.data.cameras.new("PreviewCam")); s.collection.objects.link(cam)
    cam.location = cam_loc; cam.rotation_euler = cam_rot; s.camera = cam
    sun = bpy.data.objects.new("PreviewSun", bpy.data.lights.new("PreviewSun", 'SUN')); s.collection.objects.link(sun)
    sun.rotation_euler = (0.8, 0.3, 0.5)
    s.render.engine = 'BLENDER_EEVEE'; s.render.resolution_x = 1280; s.render.resolution_y = 720
    s.render.filepath = path; bpy.ops.render.render(write_still=True)
```

# assets/blender/make_clownfish.py
# Run: /Applications/Blender.app/Contents/MacOS/Blender -b -P assets/blender/make_clownfish.py
import os, sys
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import bpy, fishlib as F

ROOT = os.path.dirname(os.path.abspath(__file__)); EXPORT = os.path.join(ROOT, "export"); os.makedirs(EXPORT, exist_ok=True)
F.reset_scene()

BODY_LEN, BODY_H, BODY_W = 11.0, 5.0, 2.2     # cm — a small, rounder fish than the tang
L = BODY_LEN / 2; PEC_ROOT_Y = BODY_W * 0.45
body = F.build_body("Clownfish", BODY_LEN, BODY_H, BODY_W, taper_z=0.5, taper_y=0.45)

def h(x):  # local half-height of the hull for fin roots
    t = max(0.0, -x / L); return (BODY_H / 2) * (1.0 - 0.5 * t * t) * (1.0 - (x / L) ** 2) ** 0.5 if abs(x) < L else 0.1

fins = [
    F.add_fin("TailFin", [(-L + 0.5, 0, 1.2), (-L - 2.8, 0, 2.4), (-L - 2.4, 0, 0), (-L - 2.8, 0, -2.4), (-L + 0.5, 0, -1.2)],
              [(0, 1, 2), (0, 2, 4), (2, 3, 4)], thickness=0.15),
    F.add_fin("DorsalFin", [(L * 0.5, 0, h(L * 0.5) * 0.7), (0, 0, h(0) * 0.7), (-L * 0.55, 0, h(-L * 0.55) * 0.7),
                            (-L * 0.55, 0, h(-L * 0.55) * 0.7 + 1.2), (0, 0, h(0) * 0.7 + 1.8), (L * 0.45, 0, h(L * 0.45) * 0.7 + 1.0)],
              [(0, 1, 4, 5), (1, 2, 3, 4)], thickness=0.15),
    F.add_fin("AnalFin", [(L * 0.05, 0, -h(L * 0.05) * 0.7), (-L * 0.55, 0, -h(-L * 0.55) * 0.7),
                          (-L * 0.55, 0, -h(-L * 0.55) * 0.7 - 1.0), (0, 0, -h(0) * 0.7 - 1.3)], [(0, 1, 2, 3)], thickness=0.15),
    F.add_fin("PecFinL", [(L * 0.35, PEC_ROOT_Y, 0.3), (L * 0.1, PEC_ROOT_Y + 2.0, -0.6), (0, PEC_ROOT_Y + 1.6, -1.1), (L * 0.25, PEC_ROOT_Y, -0.6)], [(0, 1, 2, 3)], thickness=0.15),
    F.add_fin("PecFinR", [(L * 0.35, -PEC_ROOT_Y, 0.3), (L * 0.25, -PEC_ROOT_Y, -0.6), (0, -PEC_ROOT_Y - 1.6, -1.1), (L * 0.1, -PEC_ROOT_Y - 2.0, -0.6)], [(0, 1, 2, 3)], thickness=0.15),
]
body = F.join_fins(body, fins)
F.unwrap(body)

def clownfish_color(nt, bsdf):
    # Orange body with three white vertical bars (x = head, mid, tail), each with a thin dark edge.
    geo = nt.nodes.new("ShaderNodeNewGeometry"); sep = nt.nodes.new("ShaderNodeSeparateXYZ")
    nt.links.new(geo.outputs["Position"], sep.inputs[0])     # world space; body sits at origin
    def band(center, width):
        a = nt.nodes.new("ShaderNodeMath"); a.operation = 'SUBTRACT'; a.inputs[1].default_value = center
        nt.links.new(sep.outputs["X"], a.inputs[0])
        b = nt.nodes.new("ShaderNodeMath"); b.operation = 'ABSOLUTE'; nt.links.new(a.outputs[0], b.inputs[0])
        r = nt.nodes.new("ShaderNodeMapRange"); r.inputs["From Min"].default_value = width; r.inputs["From Max"].default_value = width * 0.6
        nt.links.new(b.outputs[0], r.inputs["Value"])          # 1 inside the bar, 0 outside (inverted ramp)
        return r
    orange = (0.95, 0.35, 0.03, 1); white = (0.95, 0.95, 0.92, 1); dark = (0.03, 0.02, 0.02, 1)
    prev = None
    for cx, w in ((L * 0.55, 0.9), (0.0, 1.1), (-L * 0.6, 0.8)):
        white_m = band(cx, w); edge_m = band(cx, w + 0.35)
        mix_edge = nt.nodes.new("ShaderNodeMix"); mix_edge.data_type = 'RGBA'
        mix_edge.inputs["A"].default_value = orange; mix_edge.inputs["B"].default_value = dark
        if prev is not None: nt.links.new(prev.outputs["Result"], mix_edge.inputs["A"])
        nt.links.new(edge_m.outputs[0], mix_edge.inputs["Factor"])
        mix_white = nt.nodes.new("ShaderNodeMix"); mix_white.data_type = 'RGBA'; mix_white.inputs["B"].default_value = white
        nt.links.new(mix_edge.outputs["Result"], mix_white.inputs["A"]); nt.links.new(white_m.outputs[0], mix_white.inputs["Factor"])
        prev = mix_white
    nt.links.new(prev.outputs["Result"], bsdf.inputs["Base Color"])

F.bake_base_color(body, "M_Clownfish", clownfish_color, "T_Clownfish_BaseColor", EXPORT)
arm = F.build_rig("ClownfishRig", L, PEC_ROOT_Y, pec_z=0.3, tail_tip_x=-L - 2.6)
F.skin(body, arm)
unweighted, root_max = F.assert_rig_contract(body, arm)
F.save_and_export(body, arm, os.path.join(ROOT, "Clownfish.blend"), os.path.join(EXPORT, "Clownfish.fbx"))
F.render_preview(os.path.join(EXPORT, "preview_clownfish.png"), cam_loc=(9, -20, 4.5), cam_rot=(1.35, 0, 0.42))
print("CLOWNFISH_OK verts=%d bones=%d unweighted=%d rootMaxW=%.3f" % (len(body.data.vertices), len(arm.data.bones), unweighted, root_max))
```

void AAquariumGameMode::BeginPlay()
{
	Super::BeginPlay();
	Rng.Initialize(AssignmentSeed != 0 ? AssignmentSeed : static_cast<int32>(FDateTime::Now().GetTicks() & 0x7fffffff));
	RebuildLoadedCatalog();
}

void AAquariumGameMode::SetCatalogForTest(const TArray<FFishSpecies>& InCatalog, int32 Seed)
{
	Catalog = InCatalog;
	AssignmentSeed = Seed;
	Rng.Initialize(Seed);
	RebuildLoadedCatalog();
}

void AAquariumGameMode::RebuildLoadedCatalog()
{
	LoadedCatalogIndices.Reset();
	for (int32 i = 0; i < Catalog.Num(); ++i)
	{
		if (Catalog[i].Mesh.LoadSynchronous() != nullptr) LoadedCatalogIndices.Add(i);
		else UE_LOG(LogTemp, Warning, TEXT("Fish species %d mesh failed to load; excluded from catalog"), i);
	}
}

EBeginSessionResult AAquariumGameMode::BeginSession(const FString& RawNickname)
{
	const aquarium::BeginResult R = Session.Begin(
		std::string(TCHAR_TO_UTF8(*RawNickname)),
		static_cast<size_t>(LoadedCatalogIndices.Num()),
		[this](size_t Bound) { return static_cast<size_t>(Rng.RandRange(0, static_cast<int32>(Bound) - 1)); });
	switch (R)
	{
	case aquarium::BeginResult::InvalidNickname: return EBeginSessionResult::InvalidNickname;
	case aquarium::BeginResult::EmptyCatalog:    return EBeginSessionResult::EmptyCatalog;
	case aquarium::BeginResult::AlreadyActive:   return EBeginSessionResult::AlreadyActive;
	case aquarium::BeginResult::Ok: break;
	}
	const int32 CatalogIndex = LoadedCatalogIndices[static_cast<int32>(Session.OwnedFishIndex())];
	PlayerFishActor = SpawnPlayerFish(Catalog[CatalogIndex].Mesh.Get());
	return EBeginSessionResult::Ok;
}

AFishActor* AAquariumGameMode::SpawnPlayerFish(USkeletalMesh* Mesh)
{
	FActorSpawnParameters P; P.SpawnCollisionHandlingOverride = ESpawnActorCollisionHandlingMethod::AlwaysSpawn;
	AFishActor* Fish = GetWorld()->SpawnActor<AFishActor>(AFishActor::StaticClass(), PlaneOrigin, FRotator::ZeroRotator, P);
	Fish->bIsPlayerFish = true;
	Fish->Seed = static_cast<uint32>(Rng.RandRange(1, 1000000));
	Fish->PlaneOrigin = PlaneOrigin; Fish->PlaneHalfWidth = PlaneHalfWidth; Fish->PlaneHalfHeight = PlaneHalfHeight;
	Fish->FishMesh = Mesh;
	Fish->InitializeSwim();
	return Fish;
}

void AAquariumGameMode::EndSession()
{
	if (PlayerFishActor) { PlayerFishActor->Destroy(); PlayerFishActor = nullptr; }
	Session.End();   // drops nickname and ownership (F-14)
}

FSlateFontInfo FUiFont::Get(int32 Size)
{
	static TWeakObjectPtr<UFont> Cached;
	if (!Cached.IsValid())
	{
		UFont* Font = LoadObject<UFont>(nullptr, TEXT("/Game/UI/F_NotoSansKR.F_NotoSansKR"));
		if (!Font)
		{
			// Fallback for the "face-only" import path: compose a runtime font from the FontFace asset.
			if (UFontFace* Face = LoadObject<UFontFace>(nullptr, TEXT("/Game/UI/FF_NotoSansKR.FF_NotoSansKR")))
			{
				Font = NewObject<UFont>(GetTransientPackage(), TEXT("F_NotoSansKR_Runtime"));
				Font->FontCacheType = EFontCacheType::Runtime;
				Font->CompositeFont.DefaultTypeface.Fonts.Add(FTypefaceEntry(TEXT("Regular"), FFontData(Face)));
				Font->AddToRoot();
			}
		}
		if (!Font) { UE_LOG(LogTemp, Warning, TEXT("Noto Sans KR font asset missing; using engine default font")); return FCoreStyle::GetDefaultFontStyle("Regular", Size); }

	}

{
	const aquarium::NicknameResult R = aquarium::ValidateNickname(std::string(TCHAR_TO_UTF8(*Raw)));
	if (R.ok) return EEntryError::None;
	switch (R.error)
	{
	case aquarium::NicknameError::Empty: return EEntryError::Empty;
	case aquarium::NicknameError::TooLong: return EEntryError::TooLong;
	default: return EEntryError::InvalidCharacter;
	}
